class Component extends DCLogic { state = { pw: '', cp: '', cpError: false, done: false, toastMsg: null }; showToast(msg) { if (this._t) clearTimeout(this._t); this.setState({ toastMsg: msg }); this._t = setTimeout(() => this.setState({ toastMsg: null }), 2600); } checks() { const v = this.state.pw; return [v.length >= 8, /[A-Z]/.test(v), /[a-z]/.test(v), /[0-9]/.test(v)]; } doReset(e) { e.preventDefault(); const checks = this.checks(); const score = checks.filter(Boolean).length; if (score < 4) { this.showToast('Password does not meet the policy.'); return; } if (this.state.pw !== this.state.cp) { this.setState({ cpError: true }); return; } this.setState({ cpError: false, done: true }); } renderVals() { const checks = this.checks(); const score = checks.filter(Boolean).length; const colors = ['#DEE0E1', '#DEE0E1', '#DEE0E1', '#DEE0E1']; const activeColor = score <= 1 ? '#A32D2D' : score === 2 ? '#C97A0E' : score === 3 ? '#4D8691' : '#0F6E56'; for (let i = 0; i < score; i++) colors[i] = activeColor; const labels = ['At least 8 characters', 'One uppercase letter', 'One lowercase letter', 'One number']; const rules = labels.map((label, i) => ({ label, color: checks[i] ? '#0F6E56' : '#707A7E', weight: checks[i] ? 600 : 400 })); return { pw: this.state.pw, setPw: e => this.setState({ pw: e.target.value }), cp: this.state.cp, setCp: e => this.setState({ cp: e.target.value }), cpError: this.state.cpError, cpBorder: this.state.cpError ? '#A32D2D' : '#DBE3EC', meterBars: colors, rules, done: this.state.done, doReset: e => this.doReset(e), toastMsg: this.state.toastMsg }; } }
